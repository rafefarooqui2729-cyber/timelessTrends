/**
 * ScrollToTop helper: scrolls window to top on route change
 */
function ScrollToTop() {
  const { pathname } = useLocation();

  useEffect(() => {
    window.scrollTo(0, 0);
  }, [pathname]);

  return null;
}

export function App() {
  return (
    <BrowserRouter>
      <ScrollToTop />
      <div className="app-container">
        <Header />
        <main>
          <Routes>
            <Route path="/" element={<Home />} />

            <Route path="/collections" element={<Collections />} />

import React, { useState } from 'react';
import { useNavigate, useLocation, Navigate, Link } from 'react-router-dom';
import { Lock, Mail, ArrowRight, ShieldCheck, AlertCircle } from 'lucide-react';
import { useAdminAuth } from '../AdminAuthContext';
import '../admin.css';

export function Login() {
  const { user, login } = useAdminAuth();
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const navigate = useNavigate();
  const location = useLocation();
  const from = location.state?.from?.pathname || '/admin/dashboard';

  if (user) {
    return <Navigate to={from} replace />;
  }

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError(null);
    setLoading(true);

    try {
      await login(email, password);
      navigate(from, { replace: true });
    } catch (err) {
      setError(err.message || 'Authentication failed. Please verify credentials.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={{
      minHeight: '100vh',

            TIMELESS TRENDS

          <span style={{ fontSize: '0.72rem', letterSpacing: '0.2em', textTransform: 'uppercase', color: '#B89B62', fontWeight: 600 }}>

import React, { useState, useEffect } from 'react';
import { useParams, useNavigate, Link } from 'react-router-dom';
import {
  Watch, Save, ArrowLeft, Plus, Trash2, Upload,
  Image as ImageIcon, Check, AlertCircle, Copy, CheckCircle
} from 'lucide-react';
import { api } from '../api';
import { useAdminAuth } from '../AdminAuthContext';

export function ProductEditor() {
  const { id } = useParams();
  const isNew = !id || id === 'new';
  const navigate = useNavigate();
  const { isInventoryManager, isSuperAdmin } = useAdminAuth();

  const [categories, setCategories] = useState([]);
  const [loading, setLoading] = useState(!isNew);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState(null);
  const [success, setSuccess] = useState(null);

  // Form State
  const [name, setName] = useState('');
  const [brand, setBrand] = useState('TIMELESS TRENDS');
  const [model, setModel] = useState('');
  const [category, setCategory] = useState('');
  const [sellingPrice, setSellingPrice] = useState('');
  const [originalPrice, setOriginalPrice] = useState('');
  const [description, setDescription] = useState('');
  const [status, setStatus] = useState('draft');
  const [isNewArrival, setIsNewArrival] = useState(false);
  const [isFeatured, setIsFeatured] = useState(false);
  const [displayPriority, setDisplayPriority] = useState(0);

  // Specifications array: [{ label: string, value: string }]
  const [specs, setSpecs] = useState([
    { label: 'Case Diameter', value: '' },
    { label: 'Case Material', value: '' },
    { label: 'Movement', value: '' },
    { label: 'Water Resistance', value: '' }
  ]);

  // Variants array
  const [variants, setVariants] = useState([
    { colour: 'Default', price_override: '', min_stock: 2, stock_on_hand: 5, avg_cost: '' }
  ]);

  // Image Upload state
  const [images, setImages] = useState([]);
  const [uploadingImage, setUploadingImage] = useState(false);

  useEffect(() => {
    loadCategories();
    if (!isNew) {
      loadProduct();
    }
  }, [id]);

  async function loadCategories() {
    try {
      const data = await api('/api/admin/categories');
      setCategories(data);
      if (isNew && data.length > 0) setCategory(data[0].slug);
    } catch {}
  }

  async function loadProduct() {
    try {
      setLoading(true);
      const p = await api(`/api/admin/products/${id}`);
      setName(p.name);
      setBrand(p.brand || '');
      setModel(p.model || '');
      setCategory(p.category || '');
      setSellingPrice(p.selling_price != null ? String(p.selling_price) : '');
      setOriginalPrice(p.original_price != null ? String(p.original_price) : '');
      setDescription(p.description || '');
      setStatus(p.status || 'draft');
      setIsNewArrival(Boolean(p.is_new_arrival));
      setIsFeatured(Boolean(p.is_featured));
      setDisplayPriority(p.display_priority || 0);
      setSpecs(p.specs && p.specs.length > 0 ? p.specs : [{ label: 'Case Diameter', value: '' }]);
      setVariants(p.variants || []);
      setImages(p.images || []);
      setError(null);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }

  // Specifications helpers
  const handleAddSpec = () => {
    setSpecs([...specs, { label: '', value: '' }]);
  };

  const handleUpdateSpec = (index, field, val) => {
    const updated = [...specs];
    updated[index][field] = val;
    setSpecs(updated);
  };

  const handleRemoveSpec = (index) => {
    setSpecs(specs.filter((_, i) => i !== index));
  };

  // Variants helpers
  const handleAddVariant = () => {
    setVariants([...variants, { colour: '', price_override: '', min_stock: 2, stock_on_hand: 0, avg_cost: '' }]);
  };

  const handleUpdateVariant = (index, field, val) => {
    const updated = [...variants];
    updated[index][field] = val;
    setVariants(updated);
  };

  const handleRemoveVariant = async (index, variantId) => {
    if (!isNew && variantId) {
      if (!window.confirm('Delete this variant colour?')) return;
      try {
        await api(`/api/admin/variants/${variantId}`, { method: 'DELETE' });
        setVariants(variants.filter(v => v.id !== variantId));
        return;
      } catch (err) {
        alert(err.message);
        return;
      }
    }
    setVariants(variants.filter((_, i) => i !== index));
  };

  // Image Upload handler
  const handleImageUpload = async (e) => {
    const file = e.target.files?.[0];
    if (!file) return;

    if (isNew) {
      alert('Please save the product basic details first before uploading gallery photographs.');
      return;
    }

    try {
      setUploadingImage(true);
      const formData = new FormData();
      formData.append('image', file);
      formData.append('alt', `${name} watch photograph`);

      const updated = await api(`/api/admin/products/${id}/images`, {
        method: 'POST',
        body: formData,
        isFormData: true
      });
      setImages(updated.images || []);
      setSuccess('Image uploaded and optimized to WebP successfully!');
      setTimeout(() => setSuccess(null), 3000);
    } catch (err) {
      alert('Upload failed: ' + err.message);
    } finally {
      setUploadingImage(false);
    }
  };

  const handleDeleteImage = async (imageId) => {
    if (!window.confirm('Delete this gallery photograph?')) return;
    try {
      const updated = await api(`/api/admin/images/${imageId}`, { method: 'DELETE' });
      setImages(updated.images || []);
    } catch (err) {
      alert('Delete failed: ' + err.message);
    }
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!name.trim()) {
      alert('Product name is required.');
      return;
    }

    setSaving(true);
    setError(null);

    const validSpecs = specs.filter(s => s.label.trim() && s.value.trim());

    const payload = {
      name: name.trim(),
      brand: brand.trim() || null,
      model: model.trim() || null,
      category: category || null,
      selling_price: sellingPrice ? Number(sellingPrice) : null,
      original_price: originalPrice ? Number(originalPrice) : null,
      description: description.trim() || null,
      specs: validSpecs,
      status,
      is_new_arrival: isNewArrival,
      is_featured: isFeatured,
      display_priority: Number(displayPriority) || 0
    };

    try {
      if (isNew) {
        // Must provide at least one variant when creating
        const validVariants = variants.map(v => ({
          colour: v.colour.trim() || 'Default Finish',
          price_override: v.price_override ? Number(v.price_override) : null,
          min_stock: Number(v.min_stock) || 2,
          stock_on_hand: Number(v.stock_on_hand) || 0,
          avg_cost: v.avg_cost ? Number(v.avg_cost) : null
        }));

        const created = await api('/api/admin/products', {
          method: 'POST',
          body: { ...payload, variants: validVariants }
        });
        navigate(`/admin/products/${created.id}/edit`);
      } else {
        await api(`/api/admin/products/${id}`, {
          method: 'PUT',
          body: payload
        });
        setSuccess('Watch details updated successfully!');
        setTimeout(() => setSuccess(null), 3000);
        loadProduct();
      }
    } catch (err) {
      setError(err.message);
    } finally {
      setSaving(false);
    }
  };

  if (loading) {
    return <div style={{ color: '#888', padding: '36px 0' }}>Loading timepiece editor...</div>;
  }

  return (
    <div style={{ maxWidth: '960px', margin: '0 auto' }}>
      {/* Header */}
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '24px', flexWrap: 'wrap', gap: '16px' }}>
        <div style={{ display: 'flex', alignItems: 'center', gap: '12px' }}>
          <Link to="/admin/products" className="btn-admin-ghost" style={{ padding: '8px' }}>
            <ArrowLeft size={16} />
          </Link>
          <div>
            <h2 style={{ fontFamily: 'var(--font-serif)', fontSize: '1.8rem', color: '#fff' }}>
              {isNew ? 'Add New Timepiece' : `Edit: ${name}`}
            </h2>
            <span style={{ fontSize: '0.8rem', color: 'var(--admin-text-muted)' }}>

import React, { useState, useEffect } from 'react';
import {
  Boxes, Search, Filter, AlertTriangle, CheckCircle, ArrowDownUp,
  Download, Plus, RefreshCw, X, ShieldAlert, History, TrendingUp, DollarSign
} from 'lucide-react';
import { api } from '../api';
import { useAdminAuth } from '../AdminAuthContext';

export function InventoryManagement() {
  const { user, isSuperAdmin, isInventoryManager } = useAdminAuth();
  const [activeTab, setActiveTab] = useState('inventory'); // 'inventory' | 'movements'
  const [inventory, setInventory] = useState([]);
  const [valuation, setValuation] = useState(null);
  const [movements, setMovements] = useState([]);
  const [suppliers, setSuppliers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [searchTerm, setSearchTerm] = useState('');
  const [statusFilter, setStatusFilter] = useState('all');
  const [error, setError] = useState(null);
  const [successMsg, setSuccessMsg] = useState(null);

  // Modal states
  const [adjustModal, setAdjustModal] = useState({ open: false, item: null, newQty: '', reason: '' });
  const [receiveModal, setReceiveModal] = useState({
    open: false,
    item: null,
    supplierId: '',
    quantity: '',
    unitCost: '',
    reference: '',
    notes: ''
  });
  const [submitting, setSubmitting] = useState(false);

  const canEdit = isSuperAdmin || isInventoryManager;

  const loadData = async () => {
    setLoading(true);
    setError(null);
    try {
      const invRes = await api('/api/admin/inventory');
      setInventory(invRes.rows || []);
      setValuation(invRes.valuation || null);

      const suppRes = await api('/api/admin/suppliers').catch(() => []);
      setSuppliers(suppRes || []);

      if (activeTab === 'movements') {
        const movRes = await api('/api/admin/stock/movements');
        setMovements(movRes || []);
      }
    } catch (err) {
      setError(err.message || 'Failed to load inventory data');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    loadData();
  }, [activeTab]);

  const handleAdjustSubmit = async (e) => {
    e.preventDefault();
    if (!adjustModal.item) return;
    const newQty = parseInt(adjustModal.newQty, 10);
    if (isNaN(newQty) || newQty < 0) {
      alert('Please specify a valid non-negative quantity');
      return;
    }
    if (!adjustModal.reason.trim()) {
      alert('A reason is mandatory for manual physical adjustments');
      return;
    }

    setSubmitting(true);
    try {
      await api('/api/admin/stock/adjust', {
        method: 'POST',
        body: {
          variantId: adjustModal.item.variant_id,
          newQuantity: newQty,
          reason: adjustModal.reason.trim()
        }
      });
      setSuccessMsg(`Stock for ${adjustModal.item.product} (${adjustModal.item.colour}) updated to ${newQty}.`);
      setAdjustModal({ open: false, item: null, newQty: '', reason: '' });
      await loadData();
    } catch (err) {
      alert(err.message || 'Failed to adjust stock');
    } finally {
      setSubmitting(false);
    }
  };

  const handleReceiveSubmit = async (e) => {
    e.preventDefault();
    if (!receiveModal.item) return;
    const qty = parseInt(receiveModal.quantity, 10);
    if (isNaN(qty) || qty <= 0) {
      alert('Please enter a positive quantity to receive');
      return;
    }

    setSubmitting(true);
    try {
      const payload = {
        supplierId: receiveModal.supplierId ? parseInt(receiveModal.supplierId, 10) : null,
        reference: receiveModal.reference || null,
        notes: receiveModal.notes || null,
        items: [{
          variantId: receiveModal.item.variant_id,
          quantity: qty,
          unitCost: receiveModal.unitCost ? parseFloat(receiveModal.unitCost) : null
        }]
      };

      await api('/api/admin/stock/receive', {
        method: 'POST',
        body: payload
      });
      setSuccessMsg(`Received ${qty} units of ${receiveModal.item.product} (${receiveModal.item.colour}).`);
      setReceiveModal({
        open: false,
        item: null,
        supplierId: '',
        quantity: '',
        unitCost: '',
        reference: '',
        notes: ''
      });
      await loadData();
    } catch (err) {
      alert(err.message || 'Failed to receive stock');
    } finally {
      setSubmitting(false);
    }
  };

  const filteredInventory = inventory.filter((item) => {
    const matchesSearch =
      item.product.toLowerCase().includes(searchTerm.toLowerCase()) ||
      (item.brand && item.brand.toLowerCase().includes(searchTerm.toLowerCase())) ||
      (item.sku && item.sku.toLowerCase().includes(searchTerm.toLowerCase())) ||
      (item.colour && item.colour.toLowerCase().includes(searchTerm.toLowerCase()));

    if (!matchesSearch) return false;

    if (statusFilter === 'low_stock') return item.status === 'low_stock';
    if (statusFilter === 'out_of_stock') return item.status === 'out_of_stock';
    if (statusFilter === 'in_stock') return item.status === 'in_stock';
    return true;
  });

  const totalOnHand = inventory.reduce((acc, curr) => acc + (curr.stock_on_hand || 0), 0);
  const lowStockCount = inventory.filter((i) => i.status === 'low_stock').length;
  const outOfStockCount = inventory.filter((i) => i.status === 'out_of_stock').length;

  return (
    <div className="admin-page">
      <div className="admin-page-header">
        <div>
          <h1 className="admin-page-title">Inventory & Stock Tracking</h1>
          <p className="admin-page-subtitle">Real-time ledger of watch stock, safety thresholds, and physical counts</p>
        </div>
        <div style={{ display: 'flex', gap: '10px' }}>
          <a
            href="/api/admin/export/inventory"
            target="_blank"
            rel="noopener noreferrer"
            className="btn btn-outline"
            style={{ fontSize: '0.85rem' }}
          >
            <Download size={15} /> Export CSV
          </a>
          <button
            onClick={loadData}
            className="btn btn-secondary"
            style={{ fontSize: '0.85rem' }}
            disabled={loading}
          >
            <RefreshCw size={15} className={loading ? 'animate-spin' : ''} /> Refresh
          </button>
        </div>
      </div>

      {error && (
        <div className="admin-alert admin-alert-danger" style={{ marginBottom: '20px' }}>
          <AlertTriangle size={18} /> {error}
        </div>
      )}

      {successMsg && (
        <div className="admin-alert admin-alert-success" style={{ marginBottom: '20px' }}>
          <CheckCircle size={18} /> {successMsg}
          <button onClick={() => setSuccessMsg(null)} style={{ background: 'none', border: 'none', color: 'inherit', marginLeft: 'auto', cursor: 'pointer' }}>×</button>
        </div>
      )}

      {/* KPI Cards */}
      <div className="admin-stats-grid" style={{ marginBottom: '24px' }}>
        <div className="admin-stat-card">
          <span className="stat-label">Total Units on Hand</span>
          <span className="stat-value">{totalOnHand}</span>
          <span className="stat-detail">Across {inventory.length} watch variants</span>
        </div>
        <div className="admin-stat-card">
          <span className="stat-label">Low Stock Variants</span>
          <span className="stat-value" style={{ color: lowStockCount > 0 ? 'var(--admin-warning)' : 'inherit' }}>
            {lowStockCount}
          </span>
          <span className="stat-detail">Below safety threshold</span>
        </div>
        <div className="admin-stat-card">
          <span className="stat-label">Out of Stock</span>
          <span className="stat-value" style={{ color: outOfStockCount > 0 ? 'var(--admin-danger)' : 'inherit' }}>
            {outOfStockCount}
          </span>
          <span className="stat-detail">Unavailable for dispatch</span>
        </div>
        {valuation && isSuperAdmin && (
          <div className="admin-stat-card">
            <span className="stat-label">Total Inventory Valuation</span>
            <span className="stat-value" style={{ color: 'var(--admin-gold)' }}>
              ₹{Number(valuation.total_value_at_cost || 0).toLocaleString('en-IN')}
            </span>
            <span className="stat-detail">At average acquisition cost</span>
          </div>
        )}
      </div>

      {/* Tab Navigation */}
      <div className="admin-tabs" style={{ marginBottom: '20px' }}>
        <button
          className={`admin-tab ${activeTab === 'inventory' ? 'active' : ''}`}
          onClick={() => setActiveTab('inventory')}
        >
          <Boxes size={16} /> Current Stock Ledger ({inventory.length})
        </button>
        <button
          className={`admin-tab ${activeTab === 'movements' ? 'active' : ''}`}
          onClick={() => setActiveTab('movements')}
        >
          <History size={16} /> Stock Movement Audit ({movements.length})
        </button>
      </div>

      {activeTab === 'inventory' && (
        <div className="admin-card">
          {/* Filters Bar */}
          <div style={{ display: 'flex', gap: '12px', flexWrap: 'wrap', marginBottom: '20px', alignItems: 'center' }}>
            <div style={{ position: 'relative', flex: '1', minWidth: '220px' }}>
              <Search size={16} style={{ position: 'absolute', left: '12px', top: '50%', transform: 'translateY(-50%)', color: 'var(--admin-text-muted)' }} />
              <input
                type="text"
                placeholder="Search watch title, brand, SKU or colour..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="admin-input"
                style={{ paddingLeft: '36px' }}
              />
            </div>
            <select
              value={statusFilter}
              onChange={(e) => setStatusFilter(e.target.value)}
              className="admin-select"
              style={{ width: 'auto', minWidth: '160px' }}
            >
              <option value="all">All Stock Statuses</option>
              <option value="in_stock">In Stock (Healthy)</option>
              <option value="low_stock">Low Stock (Needs Reorder)</option>
              <option value="out_of_stock">Out of Stock</option>
            </select>
          </div>

          {/* Table */}
          <div className="admin-table-container">
            <table className="admin-table">
              <thead>
                <tr>
                  <th>Product & Variant</th>
                  <th>SKU</th>
                  <th>Stock On Hand</th>
                  <th>Reserved</th>
                  <th>Available</th>
                  <th>Min Threshold</th>
                  {isSuperAdmin && <th>Avg Cost</th>}
                  {isSuperAdmin && <th>Total Value</th>}
                  <th>Status</th>
                  {canEdit && <th style={{ textAlign: 'right' }}>Actions</th>}
                </tr>
              </thead>
              <tbody>
                {filteredInventory.length === 0 ? (
                  <tr>
                    <td colSpan={canEdit ? 10 : 8} style={{ textAlign: 'center', padding: '40px', color: 'var(--admin-text-muted)' }}>
                      No inventory records found matching your filters.
                    </td>
                  </tr>

function migrateDb(db) {
  // Apply migrations safely for existing databases
  const userCols = db.prepare("PRAGMA table_info(users)").all().map(c => c.name);
  if (!userCols.includes('is_active')) {
    try { db.exec("ALTER TABLE users ADD COLUMN is_active INTEGER NOT NULL DEFAULT 1"); } catch {}
  }

  const prodCols = db.prepare("PRAGMA table_info(products)").all().map(c => c.name);
  if (!prodCols.includes('display_priority')) {
    try { db.exec("ALTER TABLE products ADD COLUMN display_priority INTEGER NOT NULL DEFAULT 0"); } catch {}
  }
  if (!prodCols.includes('new_arrival_start')) {
    try { db.exec("ALTER TABLE products ADD COLUMN new_arrival_start TEXT"); } catch {}
  }
  if (!prodCols.includes('new_arrival_end')) {
    try { db.exec("ALTER TABLE products ADD COLUMN new_arrival_end TEXT"); } catch {}
  }

  // Seed default categories if none exist
  const catCount = db.prepare("SELECT COUNT(*) AS count FROM categories").get().count;
  if (catCount === 0) {
    const insertCat = db.prepare("INSERT INTO categories (slug, name, description, display_order, created_at) VALUES (?, ?, ?, ?, ?)");
    const now = new Date().toISOString();
    insertCat.run('everyday-classics', 'Everyday Classics', 'Timeless dials and versatile designs crafted for effortless daily wear', 1, now);
    insertCat.run('trending', 'Trending & Modern', 'Contemporary silhouettes and bold aesthetics making waves right now', 2, now);
    insertCat.run('mens', "Men's Collection", 'Sophisticated chronographs, robust bracelets, and masculine accents', 3, now);
    insertCat.run('womens', "Women's Collection", 'Delicate dials, elegant mesh bracelets, and graceful finishes', 4, now);
  }
}

function openDb(file) {
  if (file !== ':memory:') fs.mkdirSync(path.dirname(file), { recursive: true });
  const db = new Database(file);
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');
  db.exec(SCHEMA);
  migrateDb(db);
  return db;
}

// Core stock logic. EVERY change to variants.stock_on_hand goes through recordMovement(),
// so the audit trail (stock_movements) can never drift from the balances.
// Reservations (variants.reserved) are NOT stock movements: units still physically exist.
//   available (sellable now) = stock_on_hand - reserved
const { HttpError, localDate, audit } = require('./util');

function getVariant(db, id) {
  const v = db.prepare(`SELECT v.*, p.is_active AS product_active, p.name AS product_name
    FROM variants v JOIN products p ON p.id = v.product_id WHERE v.id = ?`).get(id);
  if (!v) throw new HttpError(404, `Variant ${id} not found`);
  return v;
}

function variantStatus(v) {
  if (!v.is_active || !v.product_active) return 'inactive';
  const available = v.stock_on_hand - v.reserved;
  if (available <= 0) return 'out_of_stock';
  if (available <= v.min_stock) return 'low_stock'; // "at or below the reorder level"
  return 'in_stock';
}

function recordMovement(db, { variantId, type, delta, reason = null, unitCost = null, orderId = null,
  purchaseId = null, userId = null, nonSellable = 0, occurredOn }) {
  const v = getVariant(db, variantId);
  const prev = v.stock_on_hand;
  const next = prev + delta;
  if (next < 0) throw new HttpError(409, `Not enough stock for ${v.sku}: have ${prev}, change ${delta}`);
  if (next < v.reserved) {
    throw new HttpError(409, `Would drop ${v.sku} below its ${v.reserved} reserved unit(s)`);
  }
  db.prepare('UPDATE variants SET stock_on_hand = ? WHERE id = ?').run(next, variantId);
  const now = new Date();
  db.prepare(`INSERT INTO stock_movements (occurred_at, occurred_on, product_id, variant_id, movement_type,
    quantity_change, previous_qty, new_qty, non_sellable_qty, unit_cost, reason, order_id, purchase_id, user_id)

// Supplier purchase: increases stock, updates weighted-average cost, records cash paid out.
function receiveStock(db, { supplierId = null, receivedOn, items, paidAmount, reference, notes, userId }) {
  if (!Array.isArray(items) || !items.length) throw new HttpError(400, 'items required');
  return db.transaction(() => {
    const on = receivedOn || localDate();
    let total = 0; let anyCost = false;
    for (const it of items) {
      posInt(it.quantity, 'quantity');
      if (it.unitCost != null && !(it.unitCost >= 0)) throw new HttpError(400, 'unitCost must be >= 0');
      if (it.unitCost != null) { total += it.quantity * it.unitCost; anyCost = true; }
    }
    const paid = paidAmount != null ? paidAmount : total; // default: paid in full for costed lines
    if (!(paid >= 0)) throw new HttpError(400, 'paidAmount must be >= 0');
    const pid = db.prepare(`INSERT INTO purchases (supplier_id, received_on, reference, notes, total_cost,
      paid_amount, created_by, created_at) VALUES (?,?,?,?,?,?,?,?)`)
      .run(supplierId, on, reference || null, notes || null, anyCost ? total : null, paid, userId,
        new Date().toISOString()).lastInsertRowid;
    for (const it of items) {
      const v = getVariant(db, it.variantId);
      if (it.unitCost != null) {
        const avg = (v.avg_cost == null || v.stock_on_hand <= 0) ? it.unitCost
          : (v.stock_on_hand * v.avg_cost + it.quantity * it.unitCost) / (v.stock_on_hand + it.quantity);
        db.prepare('UPDATE variants SET avg_cost = ? WHERE id = ?').run(Math.round(avg * 10000) / 10000, v.id);
      }
      recordMovement(db, { variantId: v.id, type: 'purchase_received', delta: it.quantity, unitCost: it.unitCost ?? null,
        reason: reference ? `Purchase ${reference}` : 'Stock received', purchaseId: pid, userId, occurredOn: on });
    }
    audit(db, userId, 'stock.receive', 'purchase', pid, { items });
    return pid;
  })();
}

// Manual correction to a physical count. Reason mandatory; history preserved.
function adjustStock(db, { variantId, newQuantity, reason, userId }) {
  return db.transaction(() => {
    needReason(reason);
    if (!Number.isInteger(newQuantity) || newQuantity < 0) throw new HttpError(400, 'newQuantity must be a whole number >= 0');
    const v = getVariant(db, variantId);
    const delta = newQuantity - v.stock_on_hand;
    if (delta === 0) throw new HttpError(400, 'Counted quantity equals recorded quantity; nothing to adjust');
    recordMovement(db, { variantId, type: 'manual_correction', delta, reason: String(reason).trim(), userId });
    audit(db, userId, 'stock.adjust', 'variant', variantId, { from: v.stock_on_hand, to: newQuantity, reason });
    return { previous: v.stock_on_hand, next: newQuantity };
  })();
}

function removeStock(db, { variantId, quantity, type, reason, userId }) {
  if (!['damage_loss', 'supplier_return'].includes(type)) throw new HttpError(400, 'bad type');
  return db.transaction(() => {
    posInt(quantity, 'quantity'); needReason(reason);
    recordMovement(db, { variantId, type, delta: -quantity, reason: String(reason).trim(), userId });
    audit(db, userId, `stock.${type}`, 'variant', variantId, { quantity, reason });
  })();
}

// Reconcile physical counts: only discrepancies are adjusted; each needs a reason.
function reconcile(db, { counts, userId }) {
  return db.transaction(() => {
    const out = [];
    for (const c of counts) {
      const v = getVariant(db, c.variantId);
      if (v.stock_on_hand === c.counted) { out.push({ variantId: c.variantId, changed: false }); continue; }
      adjustStock(db, { variantId: c.variantId, newQuantity: c.counted, reason: c.reason, userId });
      out.push({ variantId: c.variantId, changed: true, from: v.stock_on_hand, to: c.counted });
    }
    return out;
  })();
}

// ---- order-driven stock states (idempotent: guarded by order_items.stock_state) ----
function reserveItem(db, item) {
  if (item.stock_state === 'reserved' || item.stock_state === 'sold') return;
  const v = getVariant(db, item.variant_id);
  if (!v.is_active || !v.product_active) throw new HttpError(409, `${v.sku} is inactive`);
  if (v.stock_on_hand - v.reserved < item.quantity) {
    throw new HttpError(409, `Not enough available stock for ${v.sku} (available ${v.stock_on_hand - v.reserved}, needed ${item.quantity})`);
  }
  db.prepare('UPDATE variants SET reserved = reserved + ? WHERE id = ?').run(item.quantity, item.variant_id);
  db.prepare("UPDATE order_items SET stock_state = 'reserved' WHERE id = ?").run(item.id);
  item.stock_state = 'reserved';
}

function releaseItem(db, item) {
  if (item.stock_state !== 'reserved') return;
  db.prepare('UPDATE variants SET reserved = reserved - ? WHERE id = ?').run(item.quantity, item.variant_id);
  db.prepare("UPDATE order_items SET stock_state = 'released' WHERE id = ?").run(item.id);
  item.stock_state = 'released';
}

function sellItem(db, item, { orderId, userId, soldOn }) {
  if (item.stock_state === 'sold') return;
  reserveItem(db, item);
  const v = getVariant(db, item.variant_id);
  db.prepare('UPDATE variants SET reserved = reserved - ? WHERE id = ?').run(item.quantity, item.variant_id);
  recordMovement(db, { variantId: item.variant_id, type: 'sale', delta: -item.quantity, orderId, userId,
    reason: `Sale (order ${orderId})`, occurredOn: soldOn, unitCost: v.avg_cost });
  db.prepare("UPDATE order_items SET stock_state = 'sold', unit_cost_at_sale = ?, sold_on = ? WHERE id = ?")
    .run(v.avg_cost, soldOn, item.id);
  item.stock_state = 'sold';
}

function returnItem(db, { orderItemId, quantity, condition, reason, userId, returnedOn }) {
  return db.transaction(() => {
    posInt(quantity, 'quantity');
    if (!['sellable', 'damaged'].includes(condition)) throw new HttpError(400, "condition must be 'sellable' or 'damaged'");
    const item = db.prepare('SELECT * FROM order_items WHERE id = ?').get(orderItemId);
    if (!item) throw new HttpError(404, 'Order item not found');
    if (item.stock_state !== 'sold') throw new HttpError(409, 'Only items that were sold can be returned');
    if (quantity > item.quantity - item.returned_qty) throw new HttpError(409, 'Return quantity exceeds units sold');
    const on = returnedOn || localDate();
    const netUnit = (item.quantity * item.unit_price - item.line_discount) / item.quantity;
    const refundValue = Math.round(netUnit * quantity * 100) / 100;
    if (condition === 'sellable') {
      recordMovement(db, { variantId: item.variant_id, type: 'customer_return', delta: quantity, orderId: item.order_id,
        reason: reason || 'Customer return (sellable)', userId, occurredOn: on });
    } else {
      recordMovement(db, { variantId: item.variant_id, type: 'customer_return_damaged', delta: 0, nonSellable: quantity,
        orderId: item.order_id, reason: reason || 'Customer return (damaged, not restocked)', userId, occurredOn: on });
    }
    db.prepare('UPDATE order_items SET returned_qty = returned_qty + ? WHERE id = ?').run(quantity, item.id);
    db.prepare(`INSERT INTO returns (order_item_id, quantity, condition, returned_on, refund_value, reason, user_id, created_at)
      VALUES (?,?,?,?,?,?,?,?)`).run(item.id, quantity, condition, on, refundValue, reason || null, userId, new Date().toISOString());
    const rem = db.prepare(`SELECT SUM(quantity - returned_qty) AS left FROM order_items
      WHERE order_id = ? AND stock_state = 'sold'`).get(item.order_id).left;
    db.prepare('UPDATE orders SET status = ?, updated_at = ? WHERE id = ?')
      .run(rem === 0 ? 'returned' : 'partially_returned', new Date().toISOString(), item.order_id);
    audit(db, userId, 'order.return', 'order_item', item.id, { quantity, condition, refundValue });
    return { refundValue };
  })();
}
